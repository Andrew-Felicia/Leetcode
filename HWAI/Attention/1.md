## How to understand the problem

Each token is a node in a directed graph.

For a later token `j`, it may receive information from earlier tokens:

```text
0, 1, ..., j - 1  →  j
```

But token `j` can receive at most `c[j]` connections.

For every possible connection `i → j`, calculate an attention score `A[i][j]`. Its contribution is:

```python
A[i][j] ** 2
```

Because every contribution is non-negative and token `j` has its own independent capacity, we do not need complicated global DP:

- For each token `j`
- Calculate all scores from `0 ... j - 1`
- Select the largest `min(c[j], j)` scores
- Add them to the answer

### RMSNorm

For one vector:

```text
x = [x1, x2, ..., xd]
```

First calculate its RMS:

```python
rms = sqrt((x1*x1 + x2*x2 + ... + xd*xd) / d)
```

Then divide every value by it:

```python
normalized_x[k] = x[k] / rms
```

For example, with `d = 2`:

```text
x = [2, 2]

rms = sqrt((4 + 4) / 2) = 2

normalized_x = [1, 1]
```

Then attention between positions `i` and `j` is:

```python
attention = dot(normalized[i], normalized[j]) / sqrt(d)
score = attention * attention
```

## Solution: `1.py`

```python
import sys
import math
import heapq


def main():
    input = sys.stdin.buffer.readline

    n, d = map(int, input().split())

    vectors = []
    for _ in range(n):
        vectors.append(list(map(float, input().split())))

    capacity = list(map(int, input().split()))

    # RMSNorm
    normalized = []
    for vector in vectors:
        mean_square = sum(value * value for value in vector) / d
        rms = math.sqrt(mean_square)

        normalized.append([
            value / rms
            for value in vector
        ])

    total_score = 0.0
    scale = math.sqrt(d)

    # For every destination token j, choose its best allowed connections.
    for j in range(1, n):
        limit = min(capacity[j], j)

        if limit == 0:
            continue

        scores = []
        target = normalized[j]

        for i in range(j):
            source = normalized[i]

            dot_product = 0.0
            for feature in range(d):
                dot_product += source[feature] * target[feature]

            attention = dot_product / scale
            scores.append(attention * attention)

        if limit == j:
            total_score += sum(scores)
        else:
            total_score += sum(heapq.nlargest(limit, scores))

    # Standard half-up rounding for a non-negative result.
    answer = math.floor(total_score * 100 + 0.5)
    print(answer)


if __name__ == "__main__":
    main()
```

## Test: `test1.py`

```python
import io
import runpy
import sys
from pathlib import Path

import pytest


SOLUTION_FILE = Path(__file__).with_name("1.py")


def run_solution(test_input, monkeypatch, capsys):
    fake_stdin = io.TextIOWrapper(
        io.BytesIO(test_input.encode())
    )
    monkeypatch.setattr(sys, "stdin", fake_stdin)

    runpy.run_path(
        str(SOLUTION_FILE),
        run_name="__main__",
    )

    return capsys.readouterr().out.strip()


@pytest.mark.parametrize(
    "test_input, expected",
    [
        # Official sample 1
        (
            """\
4 2
2.0 2.0
3.0 0.0
0.0 4.0
1.0 1.0
1 2 1 3
""",
            "600",
        ),

        # Official sample 2
        (
            """\
3 2
1.0 0.0
0.0 1.0
1.0 1.0
1 1 2
""",
            "200",
        ),

        # One-dimensional vectors: normalization gives 1 or -1.
        # Squaring the attention makes both contribute 1.
        (
            """\
2 1
5.0
-4.0
1 1
""",
            "100",
        ),

        # Position 2 may choose only one connection.
        # It must choose the score 2, instead of the smaller score 1.
        (
            """\
3 2
1.0 0.0
1.0 1.0
1.0 0.0
1 1 1
""",
            "300",
        ),

        # RMSNorm removes the effect of vector magnitude.
        (
            """\
2 2
10.0 0.0
3.0 0.0
1 1
""",
            "200",
        ),

        # Attention squared score is 0.4, so integerized output is 40.
        (
            """\
2 2
1.0 0.0
1.0 2.0
1 1
""",
            "40",
        ),
    ],
)
def test_attention_routing(test_input, expected, monkeypatch, capsys):
    assert run_solution(
        test_input,
        monkeypatch,
        capsys,
    ) == expected
```

python -m pytest test1.py -v
