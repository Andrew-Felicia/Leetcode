1. Maximum sum of a contiguous sublist

This is the classic maximum subarray problem. Kadane’s algorithm solves it in O(n) time.

class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next
def max_sublist_sum(head: ListNode) -> int:
    if head is None:
        return 0
    current_sum = head.val
    max_sum = head.val
    node = head.next
    while node:
        current_sum = max(node.val, current_sum + node.val)
        max_sum = max(max_sum, current_sum)
        node = node.next
    return max_sum

Example:

# -2 -> 1 -> -3 -> 4 -> -1 -> 2 -> 1 -> -5 -> 4
head = ListNode(-2)
head.next = ListNode(1)
head.next.next = ListNode(-3)
head.next.next.next = ListNode(4)
head.next.next.next.next = ListNode(-1)
head.next.next.next.next.next = ListNode(2)
head.next.next.next.next.next.next = ListNode(1)
head.next.next.next.next.next.next.next = ListNode(-5)
head.next.next.next.next.next.next.next.next = ListNode(4)
print(max_sublist_sum(head))

Output:

6

The optimal continuous sublist is:

4 -> -1 -> 2 -> 1

and its sum is 6.

2. Convert a fraction to a decimal string with repeating part in parentheses

The key idea is to remember at which position each remainder first appeared.

If the same remainder appears again, the digits from its first occurrence are the repeating part.

def fraction_to_decimal(n: int, m: int) -> str:
    if m == 0:
        raise ZeroDivisionError("denominator cannot be zero")
    if n == 0:
        return "0"
    result = []
    # Handle sign
    if (n < 0) != (m < 0):
        result.append("-")
    n = abs(n)
    m = abs(m)
    # Integer part
    result.append(str(n // m))
    remainder = n % m
    # No decimal part
    if remainder == 0:
        return "".join(result)
    result.append(".")
    # remainder -> position in result
    seen = {}
    while remainder != 0:
        if remainder in seen:
            index = seen[remainder]
            result.insert(index, "(")
            result.append(")")
            break
        seen[remainder] = len(result)
        remainder *= 10
        result.append(str(remainder // m))
        remainder %= m
    return "".join(result)

Examples:

print(fraction_to_decimal(1, 3))
# 0.(3)
print(fraction_to_decimal(5, 4))
# 1.25
print(fraction_to_decimal(1, 6))
# 0.1(6)
print(fraction_to_decimal(2, 7))
# 0.(285714)
print(fraction_to_decimal(-50, 8))
# -6.25
print(fraction_to_decimal(1, 2))
# 0.5

One correction to the example in the question: mathematically, 1 / 3 should be "0.(3)", not "1.(3)".
